cmake_minimum_required(VERSION 3.28)
project(ElkaAudioLink VERSION 0.1.1 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "AudioLink currently targets Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX QT_NO_KEYWORDS)
set(ELKA_SDK "${CMAKE_SOURCE_DIR}/.deps" CACHE PATH "OBS SDK root")
set(ELKA_QT "${ELKA_SDK}/obs-deps-qt6-2025-07-11-x64")
set(ELKA_DEPS "${ELKA_SDK}/obs-deps-2025-07-11-x64")
list(PREPEND CMAKE_PREFIX_PATH "${ELKA_SDK}" "${ELKA_DEPS}" "${ELKA_QT}")
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network)
find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
add_library(audiolink-core STATIC src/protocol.cpp src/identity.cpp src/network.cpp src/secure-socket.cpp
  src/monitor/vban-protocol.cpp src/monitor/vban-transmitter.cpp)
target_include_directories(audiolink-core PUBLIC src src/monitor PRIVATE "${ELKA_DEPS}/include")
target_link_libraries(audiolink-core PUBLIC Qt6::Core Qt6::Network ws2_32 iphlpapi crypt32 avrt
  PRIVATE "${ELKA_DEPS}/lib/mbedcrypto.lib" "${ELKA_DEPS}/lib/mbedx509.lib" "${ELKA_DEPS}/lib/mbedtls.lib" bcrypt)
add_library(audiolink-monitor STATIC src/monitor/monitor-return.cpp)
target_link_libraries(audiolink-monitor PUBLIC audiolink-core OBS::libobs)
add_library(elka-audiolink MODULE src/plugin.cpp src/sender-ui.cpp)
target_link_libraries(elka-audiolink PRIVATE audiolink-monitor Qt6::Widgets OBS::obs-frontend-api)
add_executable(ElkaAudioLink WIN32 src/helper.cpp)
target_link_libraries(ElkaAudioLink PRIVATE audiolink-core Qt6::Widgets)
include(CTest)
if(BUILD_TESTING)
  add_executable(receiver-ui-tests tests/receiver-ui.cpp)
  target_link_libraries(receiver-ui-tests PRIVATE audiolink-core Qt6::Widgets)
  add_test(NAME receiver-tray-and-meter COMMAND receiver-ui-tests "${CMAKE_BINARY_DIR}/receiver-preview.png")
  set_tests_properties(receiver-tray-and-meter PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=windows")
  add_executable(audiolink-tests tests/integration.cpp)
  target_link_libraries(audiolink-tests PRIVATE audiolink-core)
  add_test(NAME transport-and-local-vban COMMAND audiolink-tests)
  set_tests_properties(transport-and-local-vban PROPERTIES TIMEOUT 45)
  add_executable(monitor-tests tests/monitor.cpp)
  target_link_libraries(monitor-tests PRIVATE audiolink-monitor)
  add_test(NAME obs-monitor-mix COMMAND monitor-tests)
  set_tests_properties(obs-monitor-mix PROPERTIES TIMEOUT 30)
  add_executable(plugin-smoke tests/plugin-smoke.cpp)
  target_include_directories(plugin-smoke PRIVATE .deps/frontend31)
  target_link_libraries(plugin-smoke PRIVATE audiolink-core Qt6::Widgets OBS::libobs OBS::obs-frontend-api)
  add_dependencies(plugin-smoke elka-audiolink)
  add_test(NAME actual-obs-plugin COMMAND plugin-smoke $<TARGET_FILE:elka-audiolink> "${CMAKE_SOURCE_DIR}/data" "${CMAKE_BINARY_DIR}/plugin-preview.png")
  set_tests_properties(actual-obs-plugin PROPERTIES TIMEOUT 35)
  if(EXISTS "${ELKA_SDK}/frontend32/obs-frontend-internal.hpp")
    add_executable(plugin-smoke32 tests/plugin-smoke.cpp)
    target_compile_definitions(plugin-smoke32 PRIVATE TEST_OBS32)
    target_include_directories(plugin-smoke32 PRIVATE .deps/frontend32)
    target_link_libraries(plugin-smoke32 PRIVATE audiolink-core Qt6::Widgets OBS::libobs OBS::obs-frontend-api)
    add_dependencies(plugin-smoke32 elka-audiolink)
  endif()
endif()
install(TARGETS elka-audiolink LIBRARY DESTINATION plugin/obs-plugins/64bit RUNTIME DESTINATION plugin/obs-plugins/64bit)
install(DIRECTORY data/ DESTINATION plugin/data/obs-plugins/elka-audiolink)
install(TARGETS ElkaAudioLink RUNTIME DESTINATION receiver)
